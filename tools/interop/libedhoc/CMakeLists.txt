cmake_minimum_required(VERSION 3.21)
project(coaptic_independent_edhoc_peer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(LIBEDHOC_SOURCE "" CACHE PATH "Pinned libedhoc source checkout")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LIBEDHOC_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE LIBEDHOC_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE LIBEDHOC_GIT_RESULT)
if(NOT LIBEDHOC_GIT_RESULT EQUAL 0 OR NOT LIBEDHOC_REVISION STREQUAL "0363bd3af02265e919e888b60ccffb4606b68f16")
    message(FATAL_ERROR "LIBEDHOC_SOURCE must be revision 0363bd3af02265e919e888b60ccffb4606b68f16")
endif()
if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "This host peer is qualified with GCC/Clang")
endif()
set(LIBEDHOC_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(LIBEDHOC_ENABLE_SANITIZERS OFF CACHE BOOL "" FORCE)
set(LIBEDHOC_ENABLE_FUZZING OFF CACHE BOOL "" FORCE)
set(GEN_FILES ON CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(CONFIG_LIBEDHOC_CIPHER_SUITE_0_ENABLE 0 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_CIPHER_SUITE_2_ENABLE 1 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_CIPHER_SUITE_4_ENABLE 0 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_CIPHER_SUITE_24_ENABLE 0 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_CIPHER_SUITE_PQC_1_ENABLE 0 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_MAX_NR_OF_CIPHER_SUITES 1 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_MAX_NR_OF_METHODS 1 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_MAX_LEN_OF_CONN_ID 1 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_MAX_LEN_OF_KEM_ENCAPSULATION_KEY 32 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_MAX_LEN_OF_KEM_CIPHERTEXT 32 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_MAX_LEN_OF_MAC 32 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_MAX_NR_OF_EAD_TOKENS 1 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_MAX_LEN_OF_CRED_KEY_ID 1 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_LOG_LEVEL 0 CACHE STRING "" FORCE)
set(CONFIG_LIBEDHOC_MEM_BACKEND 0 CACHE STRING "" FORCE)
add_subdirectory("${LIBEDHOC_SOURCE}" libedhoc)
include("${LIBEDHOC_SOURCE}/cmake/sources.cmake")
add_executable(libedhoc-peer peer.c ${LIBEDHOC_CIPHER_SUITE_SOURCES})
target_include_directories(libedhoc-peer PRIVATE "${LIBEDHOC_SOURCE}/library/internal")
target_link_libraries(libedhoc-peer PRIVATE libedhoc::edhoc libedhoc::cipher_suites)
target_compile_options(libedhoc-peer PRIVATE -Wall -Wextra -Werror)
target_compile_options(libedhoc PRIVATE -Wno-unused-function)
if(MINGW)
    target_link_options(libedhoc-peer PRIVATE -pthread -static)
endif()
