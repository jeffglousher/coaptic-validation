cmake_minimum_required(VERSION 3.20)
project(coaptic_libcoap_peer C)
option(LIBCOAP_QBLOCK_SANITIZE "Instrument owned dependency lifecycle tests" OFF)
if(LIBCOAP_QBLOCK_SANITIZE)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" OR WIN32)
    message(FATAL_ERROR "Q lifecycle gate requires GNU/Clang ASan+UBSan on Linux")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(NOT DEFINED LIBCOAP_SOURCE)
  message(FATAL_ERROR "Set LIBCOAP_SOURCE to pinned libcoap 851533c3cf63d16984d370ce39d586ecb3694971")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
  WORKING_DIRECTORY "${LIBCOAP_SOURCE}" OUTPUT_VARIABLE LIBCOAP_REV
  OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT LIBCOAP_REV STREQUAL "851533c3cf63d16984d370ce39d586ecb3694971")
  message(FATAL_ERROR "Unexpected libcoap source revision: ${LIBCOAP_REV}")
endif()
option(LIBCOAP_QBLOCK_PATCH "Apply the pinned protected UDP Q-Block2 continuation fix" OFF)
if(LIBCOAP_QBLOCK_PATCH)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/patches/apply_qblock_patch.py" "${LIBCOAP_SOURCE}" --apply
    OUTPUT_VARIABLE LIBCOAP_PATCH_SHA OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
else()
  execute_process(COMMAND "${GIT_EXECUTABLE}" diff --exit-code HEAD
    WORKING_DIRECTORY "${LIBCOAP_SOURCE}" COMMAND_ERROR_IS_FATAL ANY)
  set(LIBCOAP_PATCH_SHA "none")
endif()
set(ENABLE_DOCS OFF CACHE BOOL "" FORCE)
set(ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ENABLE_Q_BLOCK ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBCOAP_SOURCE}" libcoap)
add_executable(peer-libcoap peer.c)
target_link_libraries(peer-libcoap PRIVATE coap-3)
if(ENABLE_DTLS AND DTLS_BACKEND STREQUAL "openssl")
  find_package(OpenSSL REQUIRED)
  target_link_libraries(peer-libcoap PRIVATE OpenSSL::SSL)
  target_compile_definitions(peer-libcoap PRIVATE COAPTIC_OPENSSL_DIAGNOSTICS=1)
endif()
if(WIN32)
  target_link_libraries(peer-libcoap PRIVATE ws2_32)
endif()

target_compile_definitions(peer-libcoap PRIVATE COAPTIC_LIBCOAP_PATCH_SHA="${LIBCOAP_PATCH_SHA}")
if(LIBCOAP_QBLOCK_PATCH)
  add_executable(qblock-patch-test qblock_patch_test.c)
  target_include_directories(qblock-patch-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/libcoap")
  target_link_libraries(qblock-patch-test PRIVATE coap-3)
  if(NOT WIN32)
    target_link_options(qblock-patch-test PRIVATE -Wl,--wrap=coap_malloc_type)
  else()
    message(FATAL_ERROR "Q lifecycle allocation injection requires ELF --wrap")
  endif()
  if(WIN32)
    target_link_libraries(qblock-patch-test PRIVATE ws2_32)
  endif()
endif()
