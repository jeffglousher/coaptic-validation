cmake_minimum_required(VERSION 3.20)
project(coaptic_benchmark_libcoap C)
find_package(Git REQUIRED)
if(NOT DEFINED LIBCOAP_SOURCE)
  message(FATAL_ERROR "LIBCOAP_SOURCE must name pinned libcoap source")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
  WORKING_DIRECTORY "${LIBCOAP_SOURCE}" OUTPUT_VARIABLE LIBCOAP_REV
  OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT LIBCOAP_REV STREQUAL "851533c3cf63d16984d370ce39d586ecb3694971")
  message(FATAL_ERROR "Unqualified libcoap source revision")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" diff --exit-code HEAD
  WORKING_DIRECTORY "${LIBCOAP_SOURCE}" COMMAND_ERROR_IS_FATAL ANY)
set(ENABLE_DOCS OFF CACHE BOOL "" FORCE)
set(ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
option(BENCH_ENABLE_SECURITY "Build native libcoap OSCORE and DTLS fixtures" OFF)
set(BENCH_DTLS_BACKEND "mbedtls" CACHE STRING "Existing libcoap crypto backend")
set(ENABLE_DTLS ${BENCH_ENABLE_SECURITY} CACHE BOOL "" FORCE)
set(ENABLE_OSCORE ${BENCH_ENABLE_SECURITY} CACHE BOOL "" FORCE)
if(BENCH_ENABLE_SECURITY)
  set(DTLS_BACKEND ${BENCH_DTLS_BACKEND} CACHE STRING "" FORCE)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBCOAP_SOURCE}" libcoap)
if(MSVC)
  # Pinned libcoap calls POSIX strndup; keep its source unchanged on MSVC.
  target_compile_options(coap-3 PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/win32_compat.h")
endif()
add_executable(bench-libcoap server.c)
target_link_libraries(bench-libcoap PRIVATE coap-3)
if(WIN32)
  target_link_libraries(bench-libcoap PRIVATE ws2_32 bcrypt)
endif()
